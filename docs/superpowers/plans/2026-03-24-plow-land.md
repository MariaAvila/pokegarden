z# Plow Land Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the hoe Key Item so it dynamically tills `MB_SAND` tiles on a dedicated GardenMap, creating persistent berry tree planting spots using the vanilla berry system.

**Architecture:** A new `TilledPlot[32]` array in `SaveBlock1` stores each tilled spot's position and berry tree slot ID. On hoe use, a berry tree object event is spawned at the tilled tile. On map load, `RestoreTilledPlots()` re-spawns those object events and restores the tilled metatile. Berry planting/watering/harvesting reuses the vanilla system unchanged.

**Tech Stack:** C (GBA/ARM), pokeemerald-expansion v1.9.x, PoryMap (for GardenMap), mGBA (for testing)

```c
// Precondition: caller has already checked InFarm() == TRUE.
// x, y are raw map-grid coords from GetXYCoordsOneStepInFrontOfPlayer.
static bool8 CanTill(s16 x, s16 y)
{
    u8 i;
    u8 activeCount = 0;

    // Tile in front must be sand (tillable ground)
    if (MapGridGetMetatileBehaviorAt(x, y) != MB_SAND)
        return FALSE;

    // Check for duplicates and count active plots on this map
    for (i = 0; i < TILLED_PLOTS_COUNT; i++)
    {
        if (!gSaveBlock1Ptr->tilledPlots[i].active)
            continue;
        if (gSaveBlock1Ptr->tilledPlots[i].mapLayoutId != gMapHeader.mapLayoutId)
            continue;
        // Duplicate position check
        if (gSaveBlock1Ptr->tilledPlots[i].x == x &&
            gSaveBlock1Ptr->tilledPlots[i].y == y)
            return FALSE;
        activeCount++;
    }

    // Cap at 14: player(1) + follower(1) + 14 berry trees = 16 = OBJECT_EVENTS_COUNT
    return activeCount < 14;
}
```

```c
static void ItemUseOnFieldCB_Hoe(u8 taskId)
{
    s16 x, y;
    u8 i;
    u8 berryTreeId;
    u8 localId;
    u8 objectEventId;

    GetXYCoordsOneStepInFrontOfPlayer(&x, &y);

    // Find free save slot
    for (i = 0; i < TILLED_PLOTS_COUNT; i++)
    {
        if (!gSaveBlock1Ptr->tilledPlots[i].active)
            break;
    }

    // berryTreeId and localId are a direct 1-to-1 mapping with slot index
    berryTreeId = 96 + i;
    localId = TILLED_PLOT_LOCAL_ID_BASE + i;

    // Attempt spawn BEFORE writing save data (clean rollback on failure)
    objectEventId = SpawnSpecialObjectEventParameterized(
        OBJ_EVENT_GFX_BERRY_TREE,
        MOVEMENT_TYPE_BERRY_TREE_GROWTH,
        localId,
        x + MAP_OFFSET,   // SpawnSpecialObjectEventParameterized subtracts MAP_OFFSET internally
        y + MAP_OFFSET,
        3                 // elevation 3 = standard overworld walking layer
    );

    if (objectEventId == OBJECT_EVENTS_COUNT)
    {
        // Object event pool full - abort without modifying save data or metatile
        DisplayDadsAdviceCannotUseItemMessage(taskId, gTasks[taskId].tUsingRegisteredKeyItem);
        return;
    }

    // Set the berry tree ID on the spawned object event so interaction logic can find it
    gObjectEvents[objectEventId].trainerRange_berryTreeId = berryTreeId;

    // Change the metatile visually, then redraw
    MapGridSetMetatileIdAt(x, y, METATILE_GARDEN_TILLED_SOIL);
    DrawWholeMapView();

    // Write to save data
    gSaveBlock1Ptr->tilledPlots[i].x = x;
    gSaveBlock1Ptr->tilledPlots[i].y = y;
    gSaveBlock1Ptr->tilledPlots[i].mapLayoutId = gMapHeader.mapLayoutId;
    gSaveBlock1Ptr->tilledPlots[i].berryTreeId = berryTreeId;
    gSaveBlock1Ptr->tilledPlots[i].active = TRUE;

    DestroyTask(taskId);
}
```

```c
// Called on map load. Restores tilled metatiles and berry tree object events
// for the current map. Always restores the metatile even if the object event
// pool is full, so the tile stays visually tilled.
void RestoreTilledPlots(void)
{
    u8 i;
    u8 localId;
    u8 objectEventId;

    for (i = 0; i < TILLED_PLOTS_COUNT; i++)
    {
        if (!gSaveBlock1Ptr->tilledPlots[i].active)
            continue;
        if (gSaveBlock1Ptr->tilledPlots[i].mapLayoutId != gMapHeader.mapLayoutId)
            continue;

        localId = TILLED_PLOT_LOCAL_ID_BASE + i;

        // Always restore the metatile (prevents player from re-tilling the spot)
        MapGridSetMetatileIdAt(
            gSaveBlock1Ptr->tilledPlots[i].x,
            gSaveBlock1Ptr->tilledPlots[i].y,
            METATILE_GARDEN_TILLED_SOIL
        );

        objectEventId = SpawnSpecialObjectEventParameterized(
            OBJ_EVENT_GFX_BERRY_TREE,
            MOVEMENT_TYPE_BERRY_TREE_GROWTH,
            localId,
            gSaveBlock1Ptr->tilledPlots[i].x + MAP_OFFSET,
            gSaveBlock1Ptr->tilledPlots[i].y + MAP_OFFSET,
            3
        );

        if (objectEventId == OBJECT_EVENTS_COUNT)
            continue;  // Pool full: tile restored, but not interactable this session

        gObjectEvents[objectEventId].trainerRange_berryTreeId =
            gSaveBlock1Ptr->tilledPlots[i].berryTreeId;
    }
    // No DrawWholeMapView() here. RestoreTilledPlots runs inside LoadMapFromWarp,
    // before tilesets are uploaded to VRAM. The engine calls DrawWholeMapView()
    // later in the load sequence (after CopyPrimaryTilesetToVram /
    // CopySecondaryTilesetToVram complete), which will render all restored
    // metatiles correctly at that point.
}
```
